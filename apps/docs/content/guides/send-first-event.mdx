# Send Your First Event

Emitting events to Zyvan takes a single HTTP `POST` request. Once accepted, Zyvan handles fan-out to all matching destinations, handles retries, and tracks delivery attempts.

---

import (
    "context"
    "fmt"
    "os"

    "github.com/zyvan-dev/zyvan-go"
)

func main() {
    client := zyvan.NewClient(zyvan.Config{
        APIKey:    os.Getenv("ZYVAN_API_KEY"),
        ProjectID: "proj_live_99",
    })

    event, err := client.Events.Send(context.Background(), zyvan.SendEventParams{
        Type:           "payment.succeeded",
        IdempotencyKey: "pay_98234_evt_01",
        TenantID:       "tenant_enterprise_01",
        Payload: map[string]interface{}{
            "payment_id": "pi_1234567890",
            "amount":     9900,
            "currency":   "USD",
        },
    })
    if err != nil {
        panic(err)
    }

    fmt.Printf("Queued event ID: %s\n", event.ID)
}`
